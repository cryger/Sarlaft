import { Component } from '@angular/core';

import { StatCard } from '../../shared/components/stat-card/stat-card';

interface Indicador {
  titulo: string;
  valor: number;
  icono: string;
  descripcion: string;
}

interface Actividad {
  tipo: string;
  consecutivo: string;
  fecha: string;
  estado: 'Pendiente' | 'Aprobado' | 'Rechazado';
}

interface ResumenItem {
  etiqueta: string;
  valor: number;
  porcentaje: number;
}

@Component({
  selector: 'app-dashboard',
  imports: [RouterLink,StatCard],
  templateUrl: './dashboard.html',
})
export class Dashboard {
  indicadores: Indicador[] = [
    { titulo: 'Declarantes', valor: 125, icono: 'â™™', descripcion: 'Registrados en el sistema' },
    { titulo: 'Transacciones en efectivo', valor: 28, icono: '$', descripcion: 'Requieren seguimiento' },
    { titulo: 'Ausencias registradas', valor: 16, icono: 'â–£', descripcion: 'Declaraciones de ausencia' },
    { titulo: 'Reportes ROS', valor: 9, icono: 'â–¤', descripcion: 'Reportes registrados' },
  ];

  actividad: Actividad[] = [
    { tipo: 'TransacciÃ³n', consecutivo: 'TE-0001', fecha: '07/10/2026', estado: 'Pendiente' },
    { tipo: 'ROS', consecutivo: 'ROS-0002', fecha: '06/10/2026', estado: 'Pendiente' },
    { tipo: 'Ausencia', consecutivo: 'AT-0003', fecha: '05/10/2026', estado: 'Aprobado' },
    { tipo: 'TransacciÃ³n', consecutivo: 'TE-0004', fecha: '04/10/2026', estado: 'Rechazado' },
  ];

  resumen: ResumenItem[] = [
    { etiqueta: 'Transacciones en efectivo', valor: 28, porcentaje: 55 },
    { etiqueta: 'Ausencias registradas', valor: 16, porcentaje: 35 },
    { etiqueta: 'Reportes ROS', valor: 9, porcentaje: 20 },
  ];

  estilosEstado: Record<string, string> = {
    Pendiente: 'from-orange-600 to-orange-400',
    Aprobado: 'from-green-600 to-green-400',

  get pendientes(): number {

import { FormsModule } from '@angular/forms';
import { RouterLink } from '@angular/router';

import {
  DeclarantesService,
  Declarante
} from '../../../shared/services/declarantes';

@Component({
  selector: 'app-declarantes',
  standalone: true,
  imports: [
    FormsModule,
    RouterLink
  ],
  templateUrl: './declarantes.html',
  styleUrl: './declarantes.css'
})
export class DeclarantesComponent implements OnInit {

  declarantes: Declarante[] = [];

  declarantesFiltrados: Declarante[] = [];

  cargando = false;

  terminoBusqueda = '';

  error = '';


  constructor(
    private declarantesService: DeclarantesService
  ) {}


  ngOnInit(): void {
    this.cargarDeclarantes();
  }


  cargarDeclarantes(): void {

    this.cargando = true;

    this.error = '';

    this.declarantesService.listar().subscribe({

      next: (respuesta) => {

        if (respuesta.success) {

          this.declarantes = respuesta.data;

          this.declarantesFiltrados = [
            ...respuesta.data
          ];

        } else {

          this.declarantes = [];

          this.declarantesFiltrados = [];

          this.error =
            respuesta.message ??
            'No fue posible cargar los declarantes.';

        }

        this.cargando = false;

      },

      error: (error) => {

        console.error(
          'Error cargando declarantes:',
          error
        );

        this.error =
          'No fue posible conectarse con el servidor.';

        this.cargando = false;

      }

    });

  }


  buscar(): void {

    const termino =
      this.terminoBusqueda
        .trim()
        .toLowerCase();


    if (!termino) {

      this.declarantesFiltrados = [
        ...this.declarantes
      ];

      return;
    }


    this.declarantesFiltrados =
      this.declarantes.filter(declarante =>

        declarante.nombre
          .toLowerCase()
          .includes(termino)

        ||

        declarante.numero_documento
          .toLowerCase()
          .includes(termino)

        ||

        declarante.cargo
          .toLowerCase()
          .includes(termino)

        ||

        (declarante.dependencia ?? '')
          .toLowerCase()
          .includes(termino)

        ||

        (declarante.correo ?? '')
          .toLowerCase()
          .includes(termino)

      );

  }


  limpiarBusqueda(): void {

    this.terminoBusqueda = '';

    this.declarantesFiltrados = [
      ...this.declarantes
    ];

  }


  eliminar(id: number): void {

    const confirmar = window.confirm(
      'Â¿EstÃ¡ seguro de eliminar este declarante?'
    );


    if (!confirmar) {
      return;
    }


    this.declarantesService
      .eliminar(id)
      .subscribe({

        next: (respuesta) => {

          if (respuesta.success) {

            this.cargarDeclarantes();

          } else {

            this.error =
              respuesta.message ??
              'No fue posible eliminar el declarante.';

          }

        },

        error: (error) => {

          console.error(error);

          this.error =
            'OcurriÃ³ un error al eliminar el declarante.';

        }

      });

  }

}

import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

export interface Declarante {
  [x: string]: any;
  id_declarante: number;
  nombre: string;
  tipo_documento: string;
  numero_documento: string;
  cargo: string;
  id_dependencia: number;
  dependencia?: string;
  direccion?: string;
  telefono?: string;
  correo?: string;
  fecha_registro?: string;
}

export interface RespuestaApi<T> {
  success: boolean;
  message?: string;
  data: T;
}

@Injectable({
  providedIn: 'root'
})
export class DeclarantesService {

  private apiUrl = 'http://localhost/sarlaft-api';

  constructor(
    private http: HttpClient
  ) {}

  listar(): Observable<RespuestaApi<Declarante[]>> {

    return this.http.get<RespuestaApi<Declarante[]>>(
      `${this.apiUrl}/declarantes`
    );

  }

  obtener(id: number): Observable<RespuestaApi<Declarante>> {

    return this.http.get<RespuestaApi<Declarante>>(
      `${this.apiUrl}/declarantes/${id}`
    );

  }

  crear(
    declarante: Partial<Declarante>
  ): Observable<RespuestaApi<Declarante>> {

    return this.http.post<RespuestaApi<Declarante>>(
      `${this.apiUrl}/declarantes`,
      declarante
    );

  }

  actualizar(
    id: number,
    declarante: Partial<Declarante>
  ): Observable<RespuestaApi<Declarante>> {

    return this.http.put<RespuestaApi<Declarante>>(
      `${this.apiUrl}/declarantes/${id}`,
      declarante
    );

  }

  eliminar(
    id: number
  ): Observable<RespuestaApi<null>> {

    return this.http.delete<RespuestaApi<null>>(
      `${this.apiUrl}/declarantes/${id}`
    );

  }
}
